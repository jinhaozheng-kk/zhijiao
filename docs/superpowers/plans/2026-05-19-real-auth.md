> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 落地真实鉴权（手机号 OTP + 微信 H5 OAuth 骨架 + Session/JWT + tRPC），删掉 `DEV_USER_ID` 硬编码，把 `apps/web` 的 `askTutor` server action 迁到 `tutor.ask` tRPC procedure。

## ⚠️ 关键实现红线（贯穿全 plan）

1. **Redis 是注册 payload 的 source of truth**。`auth.register` 必须：在事务内 `create User + AuthIdentity`，捕获 Prisma `P2002` 唯一约束冲突 → 回滚事务 → `redis SET register:<nonce> {used:true} EX 60`（防重试）→ `Set-Cookie __Host-zj_register_token Max-Age=0`（清 cookie）→ 抛 `REGISTER_CONFLICT`。**测试必须覆盖并发 register 同 nonce 只产出 1 个 User**。
2. tRPC 错误必须用 `cause:{reason:"..."}` 传，errorFormatter 把 reason 写到 `shape.data.reason`；客户端 refresh-on-401 只匹配 `ACCESS_EXPIRED`。
3. Access JWT 永远不查 DB；refresh / logout / session 列表才查 `Session` 表。
4. `__Host-` 前缀 cookies 必须 `Secure + Path=/ + 无 Domain`；`zj_rt`（refresh）**不能**用 `__Host-` 前缀（Path=/api/auth），改普通 `HttpOnly+Secure+SameSite=Strict`。
5. Refresh 查询不带 `revokedAt IS NULL` 过滤；先查后分支（重放检测在 revoked 命中分支触发）。
6. OTP / refresh 路径限频 **fail-close**；其他 fail-open。
7. 测试中 OTP 明文 **只能** 通过 `TestSmsAdapter` 在内存中读取；不允许从 `codeHash` 反推、不允许在 procedure 返码到 response。
8. 微信 callback 是 route handler，**无 tRPC ctx**；bind 模式 userId 由 `/auth/wechat/start?mode=bind` 验 access JWT 后写 Redis state，callback 只读。
9. 服务端清 HttpOnly cookie 一律 `Set-Cookie name=; Max-Age=0; <same attrs>`；客户端 JS 不能清。

```ts
type SeedUser = {
  id: string;
  phone: string;
  name: string;
  persona: Persona;
  gradeLevel: GradeLevel | null;
  isMinor: boolean;
  roles: Role[];
};

export async function seedTestUsers(prisma: PrismaClient) {
  const users: SeedUser[] = [
    { id: "dev-student-01", phone: "13800000001", name: "测试学生",   persona: Persona.STUDENT, gradeLevel: GradeLevel.G8, isMinor: true,  roles: [] },

    { id: "dev-admin-01",   phone: "13800000005", name: "测试管理员", persona: Persona.STUDENT, gradeLevel: null,          isMinor: false, roles: [Role.ADMIN] },
  ];

  for (const u of users) {
    await prisma.user.upsert({
      where: { id: u.id },
      create: {
        id: u.id,
        name: u.name,
        persona: u.persona,
        gradeLevel: u.gradeLevel ?? undefined,
        isMinor: u.isMinor,
        roles: { create: u.roles.map((role) => ({ role })) },
        identities: {
          create: [
            {
              provider: AuthProvider.PHONE,
              providerScope: "GLOBAL",
              providerUserId: u.phone,
            },
          ],
        },
      },
      update: {
        name: u.name,
        persona: u.persona,
        gradeLevel: u.gradeLevel ?? null,
        isMinor: u.isMinor,
      },
    });

    // 角色 idempotent upsert（upsert 内的 create 在 update 分支不会重跑）
    for (const role of u.roles) {

        where: { userId_role: { userId: u.id, role } },
        create: { userId: u.id, role },
        update: {},
      });
    }
  }

  console.log(`✓ Test users seeded: ${users.length}`);
}
```

```ts
import { SignJWT, jwtVerify, errors as joseErrors } from "jose";
import type { Persona, Role } from "@zhijiao/domain";

const ACCESS_TTL = "15m";

export interface AccessClaims {
  userId: string;
  sessionId: string;
  persona: Persona;
  roles: Role[];
}

function key(secret: string) {
  return new TextEncoder().encode(secret);
}

export async function signAccess(claims: AccessClaims, secret: string): Promise<string> {
  return new SignJWT({ ...claims })
    .setProtectedHeader({ alg: "HS256" })
    .setIssuedAt()
    .setExpirationTime(ACCESS_TTL)
    .setSubject(claims.userId)
    .sign(key(secret));
}

export type VerifyResult =
  | { ok: true; claims: AccessClaims }
  | { ok: false; reason: "ACCESS_EXPIRED" | "ACCESS_INVALID" };

export async function verifyAccess(token: string, secret: string): Promise<VerifyResult> {
  try {
    const { payload } = await jwtVerify(token, key(secret), { algorithms: ["HS256"] });
    const claims: AccessClaims = {
      userId: payload.sub as string,
      sessionId: payload.sessionId as string,
      persona: payload.persona as Persona,
      roles: (payload.roles as Role[]) ?? [],
    };
    return { ok: true, claims };
  } catch (e) {
    if (e instanceof joseErrors.JWTExpired) return { ok: false, reason: "ACCESS_EXPIRED" };
    return { ok: false, reason: "ACCESS_INVALID" };
  }
}
```

```ts
export const COOKIE_AT          = "__Host-zj_at";              // access JWT，Path=/
export const COOKIE_RT          = "zj_rt";                     // refresh token，Path=/api/auth
export const COOKIE_REGISTER    = "__Host-zj_register_token";  // register nonce，Path=/
export const COOKIE_WX_STATE    = "__Host-zj_wx_state";        // wechat state，Path=/

type CookieOpts = {
  value: string;
  httpOnly?: boolean;
  secure?: boolean;
  sameSite?: "Lax" | "Strict" | "None";
  path?: string;
  maxAge?: number;   // seconds; 0 = clear
};

export function serializeCookie(name: string, opts: CookieOpts): string {
  const parts = [`${name}=${encodeURIComponent(opts.value)}`];
  if (opts.path) parts.push(`Path=${opts.path}`);
  if (opts.maxAge !== undefined) parts.push(`Max-Age=${opts.maxAge}`);
  if (opts.sameSite) parts.push(`SameSite=${opts.sameSite}`);
  if (opts.secure) parts.push("Secure");
  if (opts.httpOnly) parts.push("HttpOnly");
  return parts.join("; ");
}

export function buildAccessCookie(value: string): string {
  return serializeCookie(COOKIE_AT, { value, httpOnly:true, secure:true, sameSite:"Lax", path:"/", maxAge:900 });
}
export function buildRefreshCookie(value: string): string {
  return serializeCookie(COOKIE_RT, { value, httpOnly:true, secure:true, sameSite:"Strict", path:"/api/auth", maxAge:2592000 });
}
export function buildRegisterCookie(value: string): string {
  return serializeCookie(COOKIE_REGISTER, { value, httpOnly:true, secure:true, sameSite:"Lax", path:"/", maxAge:600 });
}
export function buildWxStateCookie(value: string): string {
  return serializeCookie(COOKIE_WX_STATE, { value, httpOnly:true, secure:true, sameSite:"Lax", path:"/", maxAge:300 });
}

/** 清 cookie：相同 name + 相同 path，Max-Age=0 */
export function clearCookie(name: string, path: string): string {
  return serializeCookie(name, { value:"", httpOnly:true, secure:true, sameSite:"Lax", path, maxAge:0 });
}
```

```ts
import type { RateLimitDecision, RateLimitStore } from "./types.js";

type Bucket = { tokens: number; updatedAt: number };

export class InMemoryRateLimitStore implements RateLimitStore {
  private buckets = new Map<string, Bucket>();

  async take(bucket: string, capacity: number, refillPerSec: number, cost: number, nowMs: number): Promise<RateLimitDecision> {
    const b = this.buckets.get(bucket) ?? { tokens: capacity, updatedAt: nowMs };
    const elapsed = (nowMs - b.updatedAt) / 1000;
    const tokens = Math.min(capacity, b.tokens + elapsed * refillPerSec);
    if (tokens >= cost) {
      const next = { tokens: tokens - cost, updatedAt: nowMs };
      this.buckets.set(bucket, next);
      return { allowed: true, retryAfterMs: 0, remaining: next.tokens };
    }
    const need = cost - tokens;
    const retryAfterMs = Math.ceil((need / refillPerSec) * 1000);
    this.buckets.set(bucket, { tokens, updatedAt: nowMs });
    return { allowed: false, retryAfterMs, remaining: tokens };
  }

  reset() { this.buckets.clear(); }
}
```

```ts
import type Redis from "ioredis";
import { RateLimitBackendDownError, type RateLimitDecision, type RateLimitStore } from "./types.js";

// KEYS[1] = bucket key
// ARGV[1] = capacity, ARGV[2] = refillPerSec, ARGV[3] = cost, ARGV[4] = nowMs
const LUA = `
local capacity = tonumber(ARGV[1])
local refill   = tonumber(ARGV[2])
local cost     = tonumber(ARGV[3])
local nowMs    = tonumber(ARGV[4])
local data = redis.call('HMGET', KEYS[1], 'tokens', 'updatedAt')
local tokens = tonumber(data[1])
local updatedAt = tonumber(data[2])
if tokens == nil then tokens = capacity; updatedAt = nowMs end
local elapsed = (nowMs - updatedAt) / 1000
tokens = math.min(capacity, tokens + elapsed * refill)
local allowed = 0
local retryAfterMs = 0
if tokens >= cost then
  tokens = tokens - cost
  allowed = 1
else
  retryAfterMs = math.ceil(((cost - tokens) / refill) * 1000)
end
redis.call('HMSET', KEYS[1], 'tokens', tokens, 'updatedAt', nowMs)
redis.call('PEXPIRE', KEYS[1], math.ceil((capacity / refill) * 1000) + 60000)
return {allowed, retryAfterMs, tostring(tokens)}
`;

export class RedisRateLimitStore implements RateLimitStore {
  constructor(private redis: Redis) {}

  async take(bucket: string, capacity: number, refillPerSec: number, cost: number, nowMs: number): Promise<RateLimitDecision> {
    try {
      const res = await this.redis.eval(
        LUA, 1, bucket,
        String(capacity), String(refillPerSec), String(cost), String(nowMs),
      ) as [number, number, string];
      return { allowed: res[0] === 1, retryAfterMs: res[1], remaining: Number(res[2]) };
    } catch (e) {
      throw new RateLimitBackendDownError(e);
    }
  }
}
```

```ts
export const RULE_REFRESH:        RateLimitRule = { prefix: "refresh:",         capacity: 6, refillPerSec: 6/60,  failMode: "open" };

import type { RateLimitStore, RateLimitDecision } from "./types.js";
import { RateLimitBackendDownError } from "./types.js";

/**
 * 按规则做一次限频判定。
 *  - fail-open：backend 抛 → 视为 {allowed:true}
 *  - fail-close：backend 抛 → 重抛 RateLimitBackendDownError
 */
export async function takeRule(
  store: RateLimitStore,
  rule: RateLimitRule,
  keySuffix: string,
  nowMs = Date.now(),
): Promise<RateLimitDecision> {
  try {
    return await store.take(rule.prefix + keySuffix, rule.capacity, rule.refillPerSec, rule.cost ?? 1, nowMs);
  } catch (e) {
    if (e instanceof RateLimitBackendDownError) {
      if (rule.failMode === "open") return { allowed: true, retryAfterMs: 0, remaining: rule.capacity };
      throw e;
    }
    throw e;
  }
}
```

```ts
import { createHash, randomInt } from "node:crypto";
import type { PrismaClient } from "@prisma/client";
import { AuthError } from "./errors.js";

const OTP_TTL_MS = 5 * 60 * 1000;
const MAX_ATTEMPTS = 5;
const CODE_LENGTH = 6;

export function generateCode(): string {
  // 100000..999999 等概率
  return String(randomInt(100_000, 1_000_000));
}

export function hashCode(code: string, phone: string, pepper: string): string {
  return createHash("sha256").update(`${code}.${phone}.${pepper}`).digest("hex");
}

export interface IssueOtpInput {
  phone: string;
  purpose: "LOGIN" | "BIND_PHONE";
  ipCreated?: string | null;
  userAgent?: string | null;
  pepper: string;
}

export async function issueOtp(prisma: PrismaClient, input: IssueOtpInput): Promise<{ code: string; otpId: string }> {
  const code = generateCode();
  const codeHash = hashCode(code, input.phone, input.pepper);
  const row = await prisma.otpRequest.create({
    data: {
      phone: input.phone,
      purpose: input.purpose,
      codeHash,
      ipCreated: input.ipCreated ?? null,
      userAgent: input.userAgent ?? null,
      expiresAt: new Date(Date.now() + OTP_TTL_MS),
    },
  });
  return { code, otpId: row.id };
}

/** 校验：失败抛 AuthError，成功返回 otpId。命中即标记 consumed */
export async function verifyOtp(prisma: PrismaClient, phone: string, code: string, pepper: string): Promise<{ otpId: string }> {
  const row = await prisma.otpRequest.findFirst({
    where: { phone, consumedAt: null, expiredAt: null, expiresAt: { gt: new Date() } },
    orderBy: { createdAt: "desc" },
  });
  if (!row) throw new AuthError("OTP_EXPIRED");

  // attempts +1（即使匹配也 +1；不影响 consumed 路径，但反爆破在 mismatch 路径生效）
  const updated = await prisma.otpRequest.update({
    where: { id: row.id },
    data: { attempts: { increment: 1 } },
  });
  if (updated.attempts >= MAX_ATTEMPTS) {
    await prisma.otpRequest.update({ where: { id: row.id }, data: { expiredAt: new Date() } });
    throw new AuthError("OTP_MAX_ATTEMPTS");
  }

  const expected = hashCode(code, phone, pepper);
  if (expected !== row.codeHash) throw new AuthError("OTP_INVALID", { remaining: MAX_ATTEMPTS - updated.attempts });

  await prisma.otpRequest.update({ where: { id: row.id }, data: { consumedAt: new Date() } });
  return { otpId: row.id };
}

export const _internal = { OTP_TTL_MS, MAX_ATTEMPTS, CODE_LENGTH };
```

```ts
import { createHash, randomBytes } from "node:crypto";
import type { PrismaClient } from "@prisma/client";
import { SessionRevokeReason } from "@prisma/client";
import { AuthError } from "./errors.js";

const REFRESH_TTL_MS = 30 * 24 * 60 * 60 * 1000;

export function hashRefresh(token: string): string {
  return createHash("sha256").update(token).digest("hex");
}

export interface DeviceMeta { ipCreated?: string | null; userAgent?: string | null; deviceLabel?: string | null; }

export async function createSession(prisma: PrismaClient, userId: string, meta: DeviceMeta) {
  const refreshToken = randomBytes(32).toString("base64url");
  const refreshTokenHash = hashRefresh(refreshToken);
  const session = await prisma.session.create({
    data: {
      userId, refreshTokenHash,
      ipCreated: meta.ipCreated ?? null,
      userAgent: meta.userAgent ?? null,
      deviceLabel: meta.deviceLabel ?? null,
      expiresAt: new Date(Date.now() + REFRESH_TTL_MS),
    },
  });
  return { sessionId: session.id, refreshToken };
}

export async function rotateSession(prisma: PrismaClient, refreshToken: string, meta: DeviceMeta) {
  const hash = hashRefresh(refreshToken);
  const current = await prisma.session.findUnique({ where: { refreshTokenHash: hash } });

  if (!current) throw new AuthError("NO_SESSION");

  if (current.revokedAt != null) {
    // 重放检测：吊销该 user 所有未失效 session
    await prisma.session.updateMany({
      where: { userId: current.userId, revokedAt: null },
      data: { revokedAt: new Date(), revokedReason: SessionRevokeReason.REUSE_DETECTED },
    });
    throw new AuthError("REUSE_DETECTED");
  }

  if (current.expiresAt <= new Date()) {
    await prisma.session.update({
      where: { id: current.id },
      data: { revokedAt: new Date(), revokedReason: SessionRevokeReason.EXPIRED },
    });
    throw new AuthError("SESSION_REVOKED");
  }

  // 事务：旧 session revoke + 新 session insert
  return prisma.$transaction(async (tx) => {
    await tx.session.update({
      where: { id: current.id },
      data: { revokedAt: new Date(), revokedReason: SessionRevokeReason.ROTATED },
    });
    const next = await tx.session.create({
      data: {
        userId: current.userId,
        refreshTokenHash: hashRefresh(refreshToken + "__placeholder"), // 占位，立刻覆盖
        ipCreated: meta.ipCreated ?? null,
        userAgent: meta.userAgent ?? null,
        deviceLabel: meta.deviceLabel ?? null,
        expiresAt: new Date(Date.now() + REFRESH_TTL_MS),
      },
    });
    const newRefresh = randomBytes(32).toString("base64url");
    await tx.session.update({
      where: { id: next.id },
      data: { refreshTokenHash: hashRefresh(newRefresh), lastUsedAt: new Date() },
    });
    return { sessionId: next.id, userId: current.userId, refreshToken: newRefresh };
  });
}

export async function revokeSession(prisma: PrismaClient, sessionId: string, reason: SessionRevokeReason) {
  await prisma.session.update({
    where: { id: sessionId },
    data: { revokedAt: new Date(), revokedReason: reason },
  });
}

export async function revokeByRefreshToken(prisma: PrismaClient, refreshToken: string, reason: SessionRevokeReason) {
  const hash = hashRefresh(refreshToken);
  await prisma.session.updateMany({
    where: { refreshTokenHash: hash, revokedAt: null },
    data: { revokedAt: new Date(), revokedReason: reason },
  });
}
```

```ts
import { randomBytes } from "node:crypto";
import type Redis from "ioredis";
import { AuthError } from "./errors.js";

export type RegisterPayload = {
  provider: "PHONE" | "WECHAT_H5";
  providerScope: string;
  providerUserId: string;
  unionId?: string;
  profile?: { nickname?: string; avatarUrl?: string };
};

const TTL_SECONDS = 600;
const KEY = (nonce: string) => `register:${nonce}`;

export async function putRegisterPayload(redis: Redis, payload: RegisterPayload): Promise<string> {
  const nonce = randomBytes(16).toString("base64url");
  await redis.set(KEY(nonce), JSON.stringify({ payload, used: false }), "EX", TTL_SECONDS);
  return nonce;
}

export async function peekRegisterPayload(redis: Redis, nonce: string): Promise<RegisterPayload> {
  const raw = await redis.get(KEY(nonce));
  if (!raw) throw new AuthError("REGISTER_TOKEN_MISSING");
  const parsed = JSON.parse(raw) as { payload: RegisterPayload; used: boolean };
  if (parsed.used) throw new AuthError("REGISTER_TOKEN_MISSING");
  return parsed.payload;
}

/** 标记已用 + 短期保留（防 race 二次提交）。失败安全：写不进也不抛 */
export async function markRegisterUsed(redis: Redis, nonce: string, payload: RegisterPayload): Promise<void> {
  await redis.set(KEY(nonce), JSON.stringify({ payload, used: true }), "EX", 60);
}

/** 失败 + 立即弃用：把 nonce 标 used，禁止重试 */
export async function abortRegister(redis: Redis, nonce: string, payload: RegisterPayload): Promise<void> {
  await markRegisterUsed(redis, nonce, payload);
}
```

```ts
import "server-only";
import type { FetchCreateContextFnOptions } from "@trpc/server/adapters/fetch";
import { prisma } from "@zhijiao/db";
import { redis } from "../lib/redis.js";
import { verifyAccess, type AccessClaims, COOKIE_AT } from "@zhijiao/auth";

export type AuthError = "ACCESS_EXPIRED" | "ACCESS_INVALID" | "NO_SESSION";

export type Ctx = {
  db: typeof prisma;
  redis: typeof redis;
  auth: AccessClaims | null;
  authError: AuthError | null;
  ip: string;
  ua: string;
  resHeaders: Headers;
  cookies: Record<string, string>;
};

function parseCookies(header: string | null): Record<string, string> {
  const out: Record<string, string> = {};
  if (!header) return out;
  for (const part of header.split(";")) {
    const idx = part.indexOf("=");
    if (idx < 0) continue;
    const k = part.slice(0, idx).trim();
    const v = decodeURIComponent(part.slice(idx + 1).trim());
    if (k) out[k] = v;
  }
  return out;
}

export async function createContext({ req, resHeaders }: FetchCreateContextFnOptions): Promise<Ctx> {
  const cookies = parseCookies(req.headers.get("cookie"));
  const at = cookies[COOKIE_AT];
  let auth: AccessClaims | null = null;
  let authError: AuthError | null = null;
  if (!at) authError = "NO_SESSION";
  else {
    const v = await verifyAccess(at, process.env.ACCESS_JWT_SECRET ?? "");
    if (v.ok) auth = v.claims; else authError = v.reason;
  }
  const ip = req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "unknown";
  const ua = req.headers.get("user-agent") || "";
  return { db: prisma, redis, auth, authError, ip, ua, resHeaders, cookies };
}
```

```ts
"use client";

import { createTRPCReact, httpBatchLink } from "@trpc/react-query";
import { TRPCClientError, type TRPCLink } from "@trpc/client";
import { observable } from "@trpc/server/observable";
import superjson from "superjson";
import type { AppRouter } from "@/server/routers/_root";

export const trpc = createTRPCReact<AppRouter>();

/** 客户端拦截器：仅 ACCESS_EXPIRED 重试一次；其它 401 → 主动调 /api/auth/logout 清 cookie + 跳 /login */
const refreshOn401Link: TRPCLink<AppRouter> = () => {
  return ({ next, op }) =>
    observable((observer) => {
      const sub = next(op).subscribe({
        next: (val) => observer.next(val),
        error: async (err) => {
          if (!(err instanceof TRPCClientError)) { observer.error(err); return; }
          const code = err.data?.code as string | undefined;
          const reason = (err.data as { reason?: string } | undefined)?.reason;
          if (code === "UNAUTHORIZED" && reason === "ACCESS_EXPIRED" && !(op.context as { _refreshed?: boolean })._refreshed) {
            try {
              const r = await fetch("/api/auth/refresh", { method: "POST", credentials: "include" });
              if (r.ok) {
                const retried = next({ ...op, context: { ...(op.context as object), _refreshed: true } });
                retried.subscribe({ next: (v) => observer.next(v), error: (e) => observer.error(e), complete: () => observer.complete() });
                return;
              }
            } catch { /* fallthrough */ }
          }
          if (code === "UNAUTHORIZED" && (reason === "SESSION_REVOKED" || reason === "REUSE_DETECTED" || reason === "ACCESS_INVALID")) {
            try { await fetch("/api/auth/logout", { method: "POST", credentials: "include" }); } catch { /* noop */ }
            if (typeof window !== "undefined") window.location.href = "/login";
          }
          if (code === "UNAUTHORIZED" && reason === "NO_SESSION") {
            if (typeof window !== "undefined") window.location.href = "/login";
          }
          observer.error(err);
        },
        complete: () => observer.complete(),
      });
      return () => sub.unsubscribe();
    });
};

export function buildTrpcClient() {
  return trpc.createClient({
    links: [
      refreshOn401Link,
      httpBatchLink({ url: "/api/trpc", transformer: superjson, fetch: (url, init) => fetch(url, { ...init, credentials: "include" }) }),
    ],
  });
}
```

```ts
import "server-only";
import type { PrismaClient } from "@prisma/client";
import { Persona, AuthProvider } from "@prisma/client";
import {
  signAccess, createSession, buildAccessCookie, buildRefreshCookie,
  type AccessClaims,
} from "@zhijiao/auth";
import { defaultRouteFor } from "@zhijiao/domain";
import { appendSetCookie } from "./cookie-utils.js";

export async function issueSessionAndCookies(opts: {
  prisma: PrismaClient;
  userId: string;
  resHeaders: Headers;
  ip?: string | null;
  ua?: string | null;
}) {
  const user = await opts.prisma.user.findUniqueOrThrow({
    where: { id: opts.userId },
    include: { roles: true },
  });
  const { sessionId, refreshToken } = await createSession(opts.prisma, opts.userId, {
    ipCreated: opts.ip ?? null, userAgent: opts.ua ?? null,
  });
  const claims: AccessClaims = {
    userId: user.id,
    sessionId,
    persona: user.persona as Persona,
    roles: user.roles.map((r) => r.role) as AccessClaims["roles"],
  };
  const access = await signAccess(claims, process.env.ACCESS_JWT_SECRET ?? "");
  appendSetCookie(opts.resHeaders, buildAccessCookie(access));
  appendSetCookie(opts.resHeaders, buildRefreshCookie(refreshToken));
  return {
    user: { id: user.id, name: user.name, persona: user.persona, gradeLevel: user.gradeLevel, isMinor: user.isMinor },
    defaultRoute: defaultRouteFor(user.persona as Persona),
  };
}

export async function findUserByPhone(prisma: PrismaClient, phone: string) {
  const id = await prisma.authIdentity.findUnique({
    where: { uniq_identity: { provider: AuthProvider.PHONE, providerScope: "GLOBAL", providerUserId: phone } },
  });
  return id ? prisma.user.findUnique({ where: { id: id.userId } }) : null;
}
```

```ts
import "server-only";
import { z } from "zod";
import { TRPCError } from "@trpc/server";
import {
  AuthError, issueOtp, verifyOtp,
  takeRule, RedisRateLimitStore, RULE_OTP_SEND_PHONE, RULE_OTP_SEND_IP, RULE_OTP_VERIFY_PH,
  putRegisterPayload, buildRegisterCookie, clearCookie, COOKIE_REGISTER,
  RateLimitBackendDownError,
} from "@zhijiao/auth";
import { buildSmsAdapter } from "@zhijiao/integrations";
import { AuthProvider } from "@prisma/client";
import { router, publicProcedure } from "../trpc.js";
import { appendSetCookie } from "../cookie-utils.js";
import { issueSessionAndCookies, findUserByPhone } from "../auth-service.js";

const PhoneSchema = z.string().regex(/^1\d{10}$/, "手机号格式不正确（11 位以 1 开头）");

function ratelimitStore(redis: import("ioredis").default) { return new RedisRateLimitStore(redis); }

function toTRPCError(reason: string, code: "BAD_REQUEST" | "TOO_MANY_REQUESTS" | "UNAUTHORIZED" | "INTERNAL_SERVER_ERROR" | "CONFLICT", extra?: object): never {
  throw new TRPCError({ code, message: reason, cause: { reason, ...extra } });
}

export const authRouter = router({
  otpRequest: publicProcedure
    .input(z.object({ phone: PhoneSchema }))
    .mutation(async ({ ctx, input }) => {
      const store = ratelimitStore(ctx.redis);
      try {
        const r1 = await takeRule(store, RULE_OTP_SEND_PHONE, input.phone);
        if (!r1.allowed) toTRPCError("OTP_SEND_TOO_FREQUENT", "TOO_MANY_REQUESTS", { retryAfterMs: r1.retryAfterMs });
        const r2 = await takeRule(store, RULE_OTP_SEND_IP, ctx.ip);
        if (!r2.allowed) toTRPCError("OTP_SEND_TOO_FREQUENT", "TOO_MANY_REQUESTS", { retryAfterMs: r2.retryAfterMs });
      } catch (e) {
        if (e instanceof RateLimitBackendDownError) toTRPCError("RATELIMIT_BACKEND_DOWN", "INTERNAL_SERVER_ERROR");
        throw e;
      }
      const { code } = await issueOtp(ctx.db, { phone: input.phone, purpose: "LOGIN", pepper: process.env.OTP_PEPPER ?? "", ipCreated: ctx.ip, userAgent: ctx.ua });
      const sms = buildSmsAdapter(process.env);
      await sms.send(input.phone, code);
      return { ok: true as const };
    }),

  otpVerify: publicProcedure
    .input(z.object({ phone: PhoneSchema, code: z.string().regex(/^\d{6}$/) }))
    .mutation(async ({ ctx, input }) => {
      const store = ratelimitStore(ctx.redis);
      try {
        const r = await takeRule(store, RULE_OTP_VERIFY_PH, input.phone);
        if (!r.allowed) toTRPCError("OTP_VERIFY_TOO_FREQUENT", "TOO_MANY_REQUESTS", { retryAfterMs: r.retryAfterMs });
      } catch (e) {
        if (e instanceof RateLimitBackendDownError) toTRPCError("RATELIMIT_BACKEND_DOWN", "INTERNAL_SERVER_ERROR");
        throw e;
      }
      try {
        await verifyOtp(ctx.db, input.phone, input.code, process.env.OTP_PEPPER ?? "");
      } catch (e) {
        if (e instanceof AuthError) toTRPCError(e.reason, "BAD_REQUEST", e.extra);
        throw e;
      }

      // 命中已注册 → issueSession
      const existing = await findUserByPhone(ctx.db, input.phone);
      if (existing) {
        const { user, defaultRoute } = await issueSessionAndCookies({
          prisma: ctx.db, userId: existing.id, resHeaders: ctx.resHeaders, ip: ctx.ip, ua: ctx.ua,
        });
        return { status: "logged_in" as const, user, defaultRoute };
      }

      // 未注册 → 写 register payload + set cookie nonce
      const nonce = await putRegisterPayload(ctx.redis, {
        provider: "PHONE", providerScope: "GLOBAL", providerUserId: input.phone,
      });
      appendSetCookie(ctx.resHeaders, buildRegisterCookie(nonce));
      return { status: "need_register" as const };
    }),
});
```

```ts
  registerInfo: publicProcedure.query(async ({ ctx }) => {
    const nonce = ctx.cookies[COOKIE_REGISTER];
    if (!nonce) throw new TRPCError({ code: "BAD_REQUEST", message: "REGISTER_TOKEN_MISSING", cause: { reason: "REGISTER_TOKEN_MISSING" } });
    try {
      const payload = await peekRegisterPayload(ctx.redis, nonce);
      return { provider: payload.provider, providerScope: payload.providerScope, suggestedName: payload.profile?.nickname ?? null };
    } catch (e) {
      // 清前端 cookie，避免反复重试
      appendSetCookie(ctx.resHeaders, clearCookie(COOKIE_REGISTER, "/"));
      throw new TRPCError({ code: "BAD_REQUEST", message: "REGISTER_TOKEN_MISSING", cause: { reason: "REGISTER_TOKEN_MISSING" } });
    }
  }),

  register: publicProcedure
    .input(RegisterInput)
    .mutation(async ({ ctx, input }) => {
      const nonce = ctx.cookies[COOKIE_REGISTER];
      if (!nonce) toTRPCError("REGISTER_TOKEN_MISSING", "BAD_REQUEST");

      let payload;
      try {
        payload = await peekRegisterPayload(ctx.redis, nonce);
      } catch (e) {
        appendSetCookie(ctx.resHeaders, clearCookie(COOKIE_REGISTER, "/"));
        toTRPCError("REGISTER_TOKEN_MISSING", "BAD_REQUEST");
      }

      if (input.persona === "STUDENT" && !input.gradeLevel) {
        // 表单错误 —— 不清 cookie，让用户改完再交（红线 #6 / spec §3.3）
        toTRPCError("REGISTER_GRADE_REQUIRED", "BAD_REQUEST");
      }

      const isMinor = deriveIsMinor(input.persona as Persona, input.gradeLevel ?? null);

      try {
        const userId = await ctx.db.$transaction(async (tx) => {
          const u = await tx.user.create({
            data: {
              persona: input.persona,
              gradeLevel: input.gradeLevel ?? undefined,
              isMinor,
              name: input.name ?? payload!.profile?.nickname ?? null,
              identities: {
                create: [{
                  provider: payload!.provider,
                  providerScope: payload!.providerScope,
                  providerUserId: payload!.providerUserId,
                  unionId: payload!.unionId,
                  metadata: payload!.profile as object | undefined,
                }],
              },
            },
          });
          return u.id;
        });

        await markRegisterUsed(ctx.redis, nonce, payload!);
        appendSetCookie(ctx.resHeaders, clearCookie(COOKIE_REGISTER, "/"));
        const { user, defaultRoute } = await issueSessionAndCookies({
          prisma: ctx.db, userId, resHeaders: ctx.resHeaders, ip: ctx.ip, ua: ctx.ua,
        });
        return { user, defaultRoute };
      } catch (e) {
        if (isPrismaUnique(e)) {
          // 并发/双击：identity 已被另一个并发请求建出来
          await abortRegister(ctx.redis, nonce, payload!);
          appendSetCookie(ctx.resHeaders, clearCookie(COOKIE_REGISTER, "/"));
          toTRPCError("REGISTER_CONFLICT", "CONFLICT");
        }
        throw e;
      }
    }),
```

```ts
export async function POST(req: Request) {
  const cookies = parseCookies(req.headers.get("cookie"));
  const rt = cookies[COOKIE_RT];
  if (!rt) return json401("NO_SESSION");

  try {
    const { sessionId, userId, refreshToken } = await rotateSession(prisma, rt, {
      ipCreated: req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ?? null,
      userAgent: req.headers.get("user-agent") ?? null,
    });
    const user = await prisma.user.findUniqueOrThrow({ where: { id: userId }, include: { roles: true } });
    const access = await signAccess(
      { userId, sessionId, persona: user.persona as Persona, roles: user.roles.map(r => r.role) as never },
      process.env.ACCESS_JWT_SECRET ?? "",
    );
    const res = NextResponse.json({ ok: true, expiresIn: 900 });
    res.headers.append("Set-Cookie", buildAccessCookie(access));
    res.headers.append("Set-Cookie", buildRefreshCookie(refreshToken));
    return res;
  } catch (e) {
    if (e instanceof AuthError) {
      const res = json401(e.reason);
      // 重放检测 / session 过期：服务端帮忙清两个 cookie
      if (e.reason === "REUSE_DETECTED" || e.reason === "SESSION_REVOKED" || e.reason === "NO_SESSION") {
        res.headers.append("Set-Cookie", clearCookie(COOKIE_AT, "/"));
        res.headers.append("Set-Cookie", clearCookie(COOKIE_RT, "/api/auth"));
      }
      return res;
    }
    throw e;
  }
}

function json401(reason: string) {
  return NextResponse.json({ ok: false, reason }, { status: 401 });
}
```

```ts
      });
      const { defaultRoute } = await issueSessionAndCookies({
        prisma: opts.prisma, userId: sibling.userId, resHeaders: opts.resHeaders, ip: opts.ip, ua: opts.ua,
      });
      return { status: "logged_in", defaultRoute };
    }
```

```ts
import { prisma } from "@zhijiao/db";
import { redis } from "@/lib/redis";
import { buildWechatH5Adapter } from "@zhijiao/integrations";
import { consumeWechatState, COOKIE_WX_STATE, clearCookie } from "@zhijiao/auth";
import { loginOrPrepareRegisterByWechat, bindWechatToUser } from "@/server/auth-service";

export const runtime = "nodejs";

export async function GET(req: Request) {
  const url = new URL(req.url);
  const code = url.searchParams.get("code");
  const state = url.searchParams.get("state");
  const cookies = parseCookies(req.headers.get("cookie"));
  const cookieState = cookies[COOKIE_WX_STATE];

  if (!code || !state || !cookieState || cookieState !== state) {
    return redirectWithClear(new URL("/login?wechat=state_mismatch", url));
  }
  const meta = await consumeWechatState(redis, state);
  if (!meta) return redirectWithClear(new URL("/login?wechat=state_mismatch", url));

  const adapter = buildWechatH5Adapter(process.env);
  let userInfo;
  try { userInfo = await adapter.exchangeCodeForUser(code); }
  catch { return redirectWithClear(new URL("/login?wechat=code_invalid", url)); }

  const resHeadersHolder = new Headers();
  if (meta.mode === "login") {
    const r = await loginOrPrepareRegisterByWechat({
      prisma, redis,
      resHeaders: resHeadersHolder,
      appId: adapter.appId,
      openid: userInfo.openid,
      unionid: userInfo.unionid,
      profile: { nickname: userInfo.nickname, avatarUrl: userInfo.avatarUrl },
      ip: req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ?? null,
      ua: req.headers.get("user-agent") ?? null,
    });
    const target = r.status === "logged_in" ? r.defaultRoute : r.redirectPath;
    const res = NextResponse.redirect(new URL(target, url));
    resHeadersHolder.getSetCookie().forEach((c) => res.headers.append("Set-Cookie", c));
    res.headers.append("Set-Cookie", clearCookie(COOKIE_WX_STATE, "/"));
    return res;
  }

  // bind
  if (!meta.userId) return redirectWithClear(new URL("/profile?bind=state_lost", url));
  const outcome = await bindWechatToUser({
    prisma, appId: adapter.appId, openid: userInfo.openid, unionid: userInfo.unionid, bindUserId: meta.userId,
  });
  const target = outcome === "ok" ? "/profile?bind=ok" : outcome === "noop" ? "/profile?bind=noop" : "/profile?bind=conflict";
  const res = NextResponse.redirect(new URL(target, url));
  res.headers.append("Set-Cookie", clearCookie(COOKIE_WX_STATE, "/"));
  return res;
}
```

```ts
import { NextResponse, type NextRequest } from "next/server";

const PROTECTED_PREFIXES = ["/profile", "/error-book", "/teacher", "/parent", "/admin"];

export function middleware(req: NextRequest) {
  const hasAt = req.cookies.has("__Host-zj_at");
  const { pathname } = req.nextUrl;
  if (PROTECTED_PREFIXES.some((p) => pathname.startsWith(p)) && !hasAt) {
    const url = req.nextUrl.clone();
    url.pathname = "/login";
    url.searchParams.set("next", pathname);
    return NextResponse.redirect(url);
  }
  return NextResponse.next();
}

export const config = {
  matcher: ["/profile/:path*", "/error-book/:path*", "/teacher/:path*", "/parent/:path*", "/admin/:path*"],
};
```

> 这个 middleware 不读 DB、不解 JWT，只看 cookie 是否存在；强校验留给 `authedProcedure`。

```tsx
"use client";

import { useState, useEffect } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import { trpc } from "@/lib/trpc-client";

export function OtpForm() {
  const router = useRouter();
  const sp = useSearchParams();
  const phone = sp.get("phone") ?? "";
  const [code, setCode] = useState("");
  const [cooldown, setCooldown] = useState(60);

  useEffect(() => {
    if (cooldown <= 0) return;
    const id = setTimeout(() => setCooldown((c) => c - 1), 1000);
    return () => clearTimeout(id);
  }, [cooldown]);

  const verify = trpc.auth.otpVerify.useMutation({
    onSuccess: (r) => {
      if (r.status === "logged_in") router.push(r.defaultRoute);
      else router.push("/register");
    },
  });
  const resend = trpc.auth.otpRequest.useMutation({ onSuccess: () => setCooldown(60) });

  return (
    <div className="space-y-3">
      <p className="text-sm text-slate-600">验证码已发到 {phone}</p>
      <input
        type="text" inputMode="numeric" maxLength={6}
        value={code} onChange={(e) => setCode(e.target.value.replace(/\D/g, ""))}
        className="border rounded w-full px-3 py-2 tracking-widest text-center"
        placeholder="6 位验证码"
        data-testid="otp-code"
      />
      <button
        disabled={verify.isPending || !/^\d{6}$/.test(code)}
        onClick={() => verify.mutate({ phone, code })}
        className="w-full rounded bg-slate-900 text-white py-2 disabled:opacity-50"
        data-testid="otp-submit"
      >
        {verify.isPending ? "校验中..." : "登录"}
```
